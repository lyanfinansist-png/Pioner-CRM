    <script type="module">
        // ================= ИМПОРТ FIREBASE =================
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.4.0/firebase-app.js";
        import { getDatabase, ref, onValue, set } from "https://www.gstatic.com/firebasejs/10.4.0/firebase-database.js";

        // ================= ⚠️ ВСТАВЬ СВОИ КЛЮЧИ СЮДА ⚠️ =================
        // Скопируй блок const firebaseConfig = {...} из нового проекта "Пионер CRM"
        const firebaseConfig = {
            apiKey: "ТВОЙ_API_KEY",
            authDomain: "crm-pioner.firebaseapp.com",
            databaseURL: "https://crm-pioner-default-rtdb.firebaseio.com",
            projectId: "crm-pioner",
            storageBucket: "crm-pioner.appspot.com",
            messagingSenderId: "ТВОЙ_SENDER_ID",
            appId: "ТВОЙ_APP_ID"
        };
        // ==================================================================

        const app = initializeApp(firebaseConfig);
        const db = getDatabase(app);
        const dbRef = ref(db, 'crm_data');

        // ================= ДАННЫЕ И СОСТОЯНИЕ =================
        const USERS = { "Администратор": "8831", "Никита": "7492", "Давид": "3185", "Денис": "9041", "Маргарита": "5628" };
        const ADMINS = ["Администратор"]; 
        let currentUser = null;

        let localData = {
            complexes: [ { id: '1', name: 'ЖК Северный Парк' }, { id: '2', name: 'ЖК Зеленый' } ],
            clients: [ { id: '101', complexId: '1', agent: 'Никита', name: 'Иванов Иван', phone: '+7 (999) 111-22-33', apartment: '45', address: 'ул. Ленина, д. 5', budget: '5 000 000', expected: '150000', paid: '30000', description: 'Ищет с ремонтом.', messages: [{author: 'Никита', text: 'Звонил клиенту', time: '12:00'}] } ]
        };

        let state = { complexId: null, clientId: null };

        // ================= СИНХРОНИЗАЦИЯ С FIREBASE =================
        onValue(dbRef, (snapshot) => {
            const data = snapshot.val();
            if (data) {
                localData = data;
            } else {
                set(dbRef, localData);
            }
            if (currentUser) {
                const activeView = document.querySelector('.view-section.active').id;
                if (activeView === 'view-complexes') renderComplexes();
                if (activeView === 'view-clients') renderClients();
                if (activeView === 'view-details') renderDetails();
            }
        });

        function saveDataToFirebase() { set(dbRef, localData); }

        // ================= АВТОРИЗАЦИЯ =================
        window.login = function() {
            const nameInput = document.getElementById('login-name').value;
            const pinInput = document.getElementById('login-pin').value.trim();
            const errorMsg = document.getElementById('error-msg');

            if (nameInput && USERS[nameInput] && USERS[nameInput] === pinInput) {
                currentUser = nameInput;
                document.getElementById('login-screen').style.display = 'none';
                document.getElementById('app-screen').style.display = 'flex';
                document.getElementById('current-user-name').textContent = currentUser;
                document.getElementById('error-msg').style.display = 'none';
                document.getElementById('login-pin').value = '';
                navTo('complexes');
            } else {
                errorMsg.style.display = 'block';
            }
        };

        window.logout = function() {
            currentUser = null;
            document.getElementById('login-name').value = '';
            document.getElementById('login-pin').value = '';
            document.getElementById('login-screen').style.display = 'flex';
            document.getElementById('app-screen').style.display = 'none';
            state = { complexId: null, clientId: null };
        };

        window.navTo = function(viewName) {
            document.querySelectorAll('.view-section').forEach(s => s.classList.remove('active'));
            document.getElementById('view-' + viewName).classList.add('active');
            
            if (viewName === 'complexes') renderComplexes();
            if (viewName === 'clients') renderClients();
            if (viewName === 'details') renderDetails();
        };

        // ================= РЕНДЕР ЭКРАНОВ =================
        function renderComplexes() {
            const list = document.getElementById('complexes-list');
            const isAdmin = ADMINS.includes(currentUser);
            document.getElementById('admin-create-complex').style.display = isAdmin ? 'flex' : 'none';

            list.innerHTML = localData.complexes.map(jk => `
                <div class="card-item" onclick="selectComplex('${jk.id}')">
                    <div><div class="title">🏢 ${jk.name}</div></div>
                    ${isAdmin ? `<button class="btn-danger" onclick="deleteComplex(event, '${jk.id}')">Удалить</button>` : ''}
                </div>
            `).join('');
        }

        window.createComplex = function() {
            if (!ADMINS.includes(currentUser)) return;
            const name = document.getElementById('new-complex-name').value.trim();
            if (!name) return;
            localData.complexes.push({ id: Date.now().toString(), name: name });
            saveDataToFirebase();
            document.getElementById('new-complex-name').value = '';
        };

        window.deleteComplex = function(event, id) {
            event.stopPropagation();
            if (!ADMINS.includes(currentUser)) return;
            if (confirm('Удалить ЖК и всех его клиентов?')) {
                localData.complexes = localData.complexes.filter(j => j.id !== id);
                localData.clients = localData.clients.filter(c => c.complexId !== id);
                saveDataToFirebase();
            }
        };

        window.selectComplex = function(id) {
            state.complexId = id;
            const jk = localData.complexes.find(j => j.id === id);
            document.getElementById('current-complex-name').textContent = jk.name;
            navTo('clients');
        };

        function renderClients() {
            const list = document.getElementById('clients-list');
            if (!state.complexId) return;
            const isAdmin = ADMINS.includes(currentUser);

            let clients = localData.clients.filter(c => c.complexId === state.complexId);
            if (!isAdmin) clients = clients.filter(c => c.agent === currentUser);

            let totalExp = 0, totalPaid = 0;
            clients.forEach(c => { totalExp += Number(c.expected) || 0; totalPaid += Number(c.paid) || 0; });
            document.getElementById('total-expected-complex').textContent = formatMoney(totalExp);
            document.getElementById('total-paid-complex').textContent = formatMoney(totalPaid);

            const agentSelect = document.getElementById('new-client-agent');
            if (!isAdmin) { agentSelect.value = currentUser; agentSelect.disabled = true; } else { agentSelect.disabled = false; }

            if (clients.length === 0) { list.innerHTML = '<div style="color: #888;">Нет клиентов</div>'; return; }

            list.innerHTML = clients.map(client => {
                const exp = Number(client.expected) || 0, paid = Number(client.paid) || 0;
                const moneyStr = (exp > 0 || paid > 0) ? ` · 💰 ${formatMoney(paid)} / ${formatMoney(exp)}` : '';
                return `
                    <div class="card-item" onclick="selectClient('${client.id}')">
                        <div>
                            <div class="title">${client.name}</div>
                            <div class="subtitle">Кв. ${client.apartment} · Тел: ${client.phone} · Агент: ${client.agent}${moneyStr}</div>
                        </div>
                    </div>
                `;
            }).join('');
        }

        window.createClient = function() {
            if (!state.complexId) return;
            const name = document.getElementById('new-client-name').value.trim();
            const phone = document.getElementById('new-client-phone').value.trim();
            const apt = document.getElementById('new-client-apt').value.trim();
            const agent = document.getElementById('new-client-agent').value;
            if (!name || !phone || !apt) { alert('Заполните поля!'); return; }

            localData.clients.push({
                id: Date.now().toString(), complexId: state.complexId, agent: agent,
                name: name, phone: phone, apartment: apt, address: '', budget: '', description: '', expected: '', paid: '', messages: []
            });
            saveDataToFirebase();
            document.getElementById('new-client-name').value = '';
            document.getElementById('new-client-phone').value = '';
            document.getElementById('new-client-apt').value = '';
        };

        window.selectClient = function(id) {
            state.clientId = id;
            navTo('details');
        };

        function renderDetails() {
            const client = localData.clients.find(c => c.id === state.clientId);
            if (!client) return;
            const isAdmin = ADMINS.includes(currentUser);

            document.getElementById('detail-header-name').textContent = client.name;
            document.getElementById('fio').value = client.name || '';
            document.getElementById('apartment').value = client.apartment || '';
            document.getElementById('phone').value = client.phone || '';
            document.getElementById('address').value = client.address || '';
            document.getElementById('budget').value = client.budget || '';
            document.getElementById('input-expected').value = client.expected || '';
            document.getElementById('input-paid').value = client.paid || '';
            document.getElementById('description').value = client.description || '';
            document.getElementById('client-agent').value = client.agent || '';

            const phoneLink = client.phone ? client.phone.replace(/[^\d+]/g, '') : '';
            document.getElementById('call-link').href = 'tel:' + phoneLink;

            const adminFields = document.querySelectorAll('.admin-only');
            adminFields.forEach(field => { field.disabled = !isAdmin; field.style.cursor = isAdmin ? 'text' : 'not-allowed'; });
            document.getElementById('admin-only-label').textContent = isAdmin ? '' : '(только админ)';
            document.getElementById('btn-delete-client').style.display = isAdmin ? 'block' : 'none';

            renderChat(client);
            updateTotals();
        }

        function formatMoney(amount) { return amount.toLocaleString('ru-RU') + ' ₽'; }

        window.updateTotals = function() {
            const expected = Number(document.getElementById('input-expected').value) || 0;
            const paid = Number(document.getElementById('input-paid').value) || 0;
            document.getElementById('total-expected').textContent = formatMoney(expected);
            document.getElementById('total-paid').textContent = formatMoney(paid);
            document.getElementById('sidebar-total').textContent = `Итого: ${formatMoney(paid)} / ${formatMoney(expected)}`;
        };

        window.saveClientData = function() {
            const client = localData.clients.find(c => c.id === state.clientId);
            if (!client) return;
            const isAdmin = ADMINS.includes(currentUser);

            client.name = document.getElementById('fio').value;
            client.apartment = document.getElementById('apartment').value;
            client.phone = document.getElementById('phone').value;
            client.address = document.getElementById('address').value;
            client.budget = document.getElementById('budget').value;
            client.description = document.getElementById('description').value;
            if (isAdmin) {
                client.expected = document.getElementById('input-expected').value;
                client.paid = document.getElementById('input-paid').value;
            }

            saveDataToFirebase();
            document.getElementById('detail-header-name').textContent = client.name;
            const phoneLink = client.phone ? client.phone.replace(/[^\d+]/g, '') : '';
            document.getElementById('call-link').href = 'tel:' + phoneLink;
            alert('✅ Данные сохранены в облако!');
        };

        window.deleteClient = function() {
            if (!ADMINS.includes(currentUser)) return;
            if (confirm('Удалить этого клиента?')) {
                localData.clients = localData.clients.filter(c => c.id !== state.clientId);
                saveDataToFirebase();
                navTo('clients');
            }
        };

        window.copyPhone = function() {
            const phone = document.getElementById('phone').value;
            if (!phone) return;
            navigator.clipboard.writeText(phone).then(() => alert('Номер скопирован: ' + phone)).catch(()=>alert('Ошибка'));
        };

        // ================= ЧАТ =================
        function renderChat(client) {
            const container = document.getElementById('chat-history');
            if (!client.messages || client.messages.length === 0) {
                container.innerHTML = '<div style="color: #555; text-align: center; margin-top: 20px;">Нет записей.</div>';
                return;
            }
            container.innerHTML = client.messages.map((msg, index) => {
                const textWithMentions = msg.text.replace(/(@[А-Яа-яA-Za-z0-9_]+)/g, '<span class="mention">$1</span>');
                const editBtn = msg.author === currentUser ? `<button class="btn-edit-msg" onclick="editMessage(${index})">✏️ Ред.</button>` : '';
                return `
                    <div class="chat-message">
                        <div class="chat-meta"><span>${msg.author} · ${msg.time}</span>${editBtn}</div>
                        <div class="chat-text">${textWithMentions}</div>
                    </div>
                `;
            }).join('');
            container.scrollTop = container.scrollHeight;
        }

        window.sendMessage = function() {
            const input = document.getElementById('chat-input');
            const text = input.value.trim();
            if (!text) return;
            const client = localData.clients.find(c => c.id === state.clientId);
            if (!client.messages) client.messages = [];
            client.messages.push({ author: currentUser, text: text, time: new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' }) });
            saveDataToFirebase();
            input.value = '';
        };

        window.editMessage = function(index) {
            const client = localData.clients.find(c => c.id === state.clientId);
            if (!client || !client.messages[index]) return;
            const currentText = client.messages[index].text;
            const newText = prompt("Редактировать сообщение:", currentText);
            if (newText !== null) {
                if (newText.trim() === "") {
                    if(confirm("Текст пустой. Удалить сообщение?")) client.messages.splice(index, 1);
                } else {
                    client.messages[index].text = newText.trim();
                }
                saveDataToFirebase();
            }
        };
